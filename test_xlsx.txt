import re, py_compile, shutil, sys

f = "/opt/conda/envs/DWS-GPU/lib/python3.11/site-packages/flashinfer/comm/fd_exchange.py"

# 1. Backup
shutil.copy(f, f + ".bak")

# 2. Lire
src = open(f, encoding="utf-8").read()

# 3. Déjà patché ?
if "from __future__ import annotations" in src.split("\n\n")[0] or \
   src.lstrip().startswith("from __future__ import annotations"):
    print("Déjà patché ✅")
else:
    lines = src.split("\n")
    # Insérer après le shebang/docstring éventuel, avant le 1er import
    insert_at = 0
    for i, ln in enumerate(lines):
        s = ln.strip()
        if s.startswith("#") or s == "" or s.startswith('"""') or s.startswith("'''"):
            continue
        insert_at = i
        break
    lines.insert(insert_at, "from __future__ import annotations")
    open(f, "w", encoding="utf-8").write("\n".join(lines))
    print(f"Ligne insérée à la position {insert_at} ✅")

# 4. Vérifier que le fichier compile
try:
    py_compile.compile(f, doraise=True)
    print("Compilation Python OK ✅")
except Exception as e:
    print("ÉCHEC compilation, restauration backup ❌:", e)
    shutil.copy(f + ".bak", f)
    sys.exit(1)
