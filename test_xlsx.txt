def generate_forfait(result):
    if not TEMPLATE_FORFAIT.exists():
        print(f'❌ Template introuvable : {TEMPLATE_FORFAIT}'); return None
    out = FORFAIT_OUTPUT_DIR / f'FORFAIT Cas 1_{_clean(result.get("entreprise"))}_{result.get("exercice") or "XXXX"}.xlsx'
    shutil.copy2(TEMPLATE_FORFAIT, out)
    wb = load_workbook(out)
    a, p, t = result.get('ACTIF') or {}, result.get('PASSIF') or {}, result.get('TCR') or {}
    n_wr = 0

    # ACTIF
    ws = wb['Saisie actif']
    ws.cell(row=2, column=2, value='Fiscal')
    if result.get('exercice'):
        ws.cell(row=4, column=2, value=result['exercice'])
    for key, row in ACTIF_ROWS.items():
        for colkey, col in ACTIF_COLS.items():
            v = to_kdzd(a.get(f'{key}_{colkey}'))
            if v is not None: ws.cell(row=row, column=col, value=v); n_wr += 1

    # PASSIF
    ws = wb['Saisie passif']
    for key, row in PASSIF_ROWS.items():
        for colkey, col in PASSIF_COLS.items():
            v = to_kdzd(p.get(f'{key}_{colkey}'))
            if v is not None: ws.cell(row=row, column=col, value=v); n_wr += 1

    # TCR (ventilation DEBIT/CREDIT)
    ws = wb['Saisie TCR']
    for key, row in TCR_ROWS.items():
        for suf, cd, cc in [('n', TCR_COLS['deb_n'], TCR_COLS['cred_n']),
                            ('n1', TCR_COLS['deb_n1'], TCR_COLS['cred_n1'])]:
            v = t.get(f'{key}_{suf}')
            if v is None: continue
            av = round(abs(v), 2)
            if key in TCR_CREDIT:
                ws.cell(row=row, column=cc if v >= 0 else cd, value=av)
            elif key in TCR_DEBIT:
                ws.cell(row=row, column=cd if v >= 0 else cc, value=av)
            else:  # résultat variable
                ws.cell(row=row, column=cc if v >= 0 else cd, value=av)
            n_wr += 1

    wb.save(out); wb.close()
    print(f'✅ {out.name} | {n_wr} cellules')
    return out
