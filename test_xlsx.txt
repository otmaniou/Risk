import re, shutil
from openpyxl import load_workbook

def to_kdzd(v):
    return None if v is None else round(v / 1000.0, 2)

def _clean_name(s):
    return re.sub(r'[^A-Za-z0-9 _-]', '', s or 'ENTREPRISE').strip()

def _norm(s):
    return re.sub(r'[^a-z0-9]', '', str(s or '').lower())

def find_row(ws, patterns, max_col=4, max_row=90):
    for r in range(1, max_row + 1):
        for c in range(1, max_col + 1):
            v = _norm(ws.cell(row=r, column=c).value)
            if not v: continue
            for pat in patterns:
                if _norm(pat) and _norm(pat) in v:
                    return r
    return None

def detect_passif_cols(ws):
    """Trouve les colonnes N et N-1 via les en-têtes de dates 12/20xx."""
    for r in range(1, 12):
        dates = [c for c in range(1, 14)
                 if ws.cell(row=r, column=c).value
                 and re.search(r'\d{2}/20\d{2}', str(ws.cell(row=r, column=c).value))]
        if len(dates) >= 2:
            return dates[0], dates[1]
    return None, None

def generate_forfait(result):
    if not TEMPLATE_FORFAIT.exists():
        print(f'❌ Template introuvable : {TEMPLATE_FORFAIT}'); return None
    out = FORFAIT_OUTPUT_DIR / f'FORFAIT Cas 1_{_clean_name(result.get("entreprise"))}_{result.get("exercice")}.xlsx'
    shutil.copy2(TEMPLATE_FORFAIT, out)
    wb = load_workbook(out)
    a, p, t = result.get('ACTIF') or {}, result.get('PASSIF') or {}, result.get('TCR') or {}
    n_wr = 0

    # ── ACTIF (inchangé, fonctionne) ──
    # ... votre bloc actif existant ...

    # ── PASSIF (corrigé : colonnes auto-détectées) ──
    if 'Saisie passif' in wb.sheetnames:
        ws = wb['Saisie passif']
        col_n, col_n1 = detect_passif_cols(ws)
        if col_n is None:
            col_n, col_n1 = 2, 3   # repli
        for key, pats in PASSIF_ROWS.items():
            r = find_row(ws, pats, max_col=col_n)   # libellé à gauche des colonnes de données
            if r is None:
                continue
            v = to_kdzd(p.get(f'{key}_n'))
            if v is not None: ws.cell(row=r, column=col_n, value=v); n_wr += 1
            v = to_kdzd(p.get(f'{key}_n1'))
            if v is not None: ws.cell(row=r, column=col_n1, value=v); n_wr += 1

    # ── TCR (inchangé, fonctionne) ──
    # ... votre bloc tcr existant ...

    wb.save(out); wb.close()
    print(f'✅ {out.name} | {n_wr} cellules')
    return out
