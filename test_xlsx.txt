# ─── FONCTIONS ──────────────────────────────────────────────
def to_kdzd(v):
    """Dinars → KDZD (÷1000, arrondi 2 décimales)."""
    if v is None:
        return None
    return round(v / 1000.0, 2)

def _clean_name(s):
    """Nettoie le nom d'entreprise pour le nom de fichier."""
    return re.sub(r'[^A-Za-z0-9 _-]', '', s or 'ENTREPRISE').strip()

def generate_forfait(result):
    """Génère un fichier FORFAIT à partir des données extraites."""
    
    # Vérifier le template
    if not TEMPLATE_FORFAIT.exists():
        print(f"❌ Template introuvable : {TEMPLATE_FORFAIT}")
        return None
    
    if TEMPLATE_FORFAIT.suffix != '.xlsx':
        print(f"❌ Le template doit être en .xlsx : {TEMPLATE_FORFAIT}")
        print(f"   Extension actuelle : {TEMPLATE_FORFAIT.suffix}")
        return None
    
    entreprise = _clean_name(result.get('entreprise'))
    annee = result.get('exercice') or 'XXXX'
    out_path = FORFAIT_OUTPUT_DIR / f'FORFAIT Cas 1_{entreprise}_{annee}.xlsx'
    
    try:
        # Copier le template
        shutil.copy2(TEMPLATE_FORFAIT, out_path)
        
        # Ouvrir la copie avec openpyxl
        wb = load_workbook(out_path)
        
        actif = result.get('ACTIF') or {}
        passif = result.get('PASSIF') or {}
        tcr = result.get('TCR') or {}
        decl = result.get('DECL') or {}
        
        n_filled = 0
        
        # ── Saisie actif ────────────────────────────────────
        if 'Saisie actif' in wb.sheetnames:
            ws = wb['Saisie actif']
            
            # En-tête : Nature bilan (colonne O, ligne 2)
            ws.cell(row=2, column=15, value='Fiscal')
            
            # Date d'arrêté N (colonne O, ligne 4)
            exerc = result.get('exercice')
            if exerc:
                ws.cell(row=4, column=15, value=exerc)
            
            # Postes
            for key, row in FORFAIT_ACTIF_COORDS.items():
                # N : brut, amort, net
                v_brut = to_kdzd(actif.get(f'{key}_brut'))
                v_amort = to_kdzd(actif.get(f'{key}_amort'))
                v_net = to_kdzd(actif.get(f'{key}_n'))
                
                if v_brut is not None:
                    ws.cell(row=row, column=ACTIF_COLS['n']['brut'], value=v_brut)
                    n_filled += 1
                if v_amort is not None:
                    ws.cell(row=row, column=ACTIF_COLS['n']['amort'], value=v_amort)
                    n_filled += 1
                if v_net is not None:
                    ws.cell(row=row, column=ACTIF_COLS['n']['net'], value=v_net)
                    n_filled += 1
                
                # N-1 : net uniquement
                v_net_n1 = to_kdzd(actif.get(f'{key}_n1'))
                if v_net_n1 is not None:
                    ws.cell(row=row, column=ACTIF_COLS['n1']['net'], value=v_net_n1)
                    n_filled += 1
        else:
            print(f"⚠️ Feuille 'Saisie actif' absente")
        
        # ── Saisie passif ───────────────────────────────────
        if 'Saisie passif' in wb.sheetnames:
            ws = wb['Saisie passif']
            
            for key, row in FORFAIT_PASSIF_COORDS.items():
                v_n = to_kdzd(passif.get(f'{key}_n'))
                v_n1 = to_kdzd(passif.get(f'{key}_n1'))
                
                if v_n is not None:
                    ws.cell(row=row, column=PASSIF_COLS['n'], value=v_n)
                    n_filled += 1
                if v_n1 is not None:
                    ws.cell(row=row, column=PASSIF_COLS['n1'], value=v_n1)
                    n_filled += 1
        else:
            print(f"⚠️ Feuille 'Saisie passif' absente")
        
        # ── Saisie TCR ──────────────────────────────────────
        if 'Saisie TCR' in wb.sheetnames:
            ws = wb['Saisie TCR']
            
            for key, row in FORFAIT_TCR_COORDS.items():
                v_deb_n = to_kdzd(tcr.get(f'{key}_debit_n'))
                v_cred_n = to_kdzd(tcr.get(f'{key}_credit_n'))
                v_deb_n1 = to_kdzd(tcr.get(f'{key}_debit_n1'))
                v_cred_n1 = to_kdzd(tcr.get(f'{key}_credit_n1'))
                
                if v_deb_n is not None:
                    ws.cell(row=row, column=TCR_COLS['n']['debit'], value=v_deb_n)
                    n_filled += 1
                if v_cred_n is not None:
                    ws.cell(row=row, column=TCR_COLS['n']['credit'], value=v_cred_n)
                    n_filled += 1
                if v_deb_n1 is not None:
                    ws.cell(row=row, column=TCR_COLS['n1']['debit'], value=v_deb_n1)
                    n_filled += 1
                if v_cred_n1 is not None:
                    ws.cell(row=row, column=TCR_COLS['n1']['credit'], value=v_cred_n1)
                    n_filled += 1
        else:
            print(f"⚠️ Feuille 'Saisie TCR' absente")
        
        # Sauvegarder
        wb.save(out_path)
        wb.close()
        
        print(f"✅ FORFAIT : {out_path.name} | {n_filled} cellules remplies")
        return out_path
        
    except Exception as e:
        print(f"❌ Erreur génération FORFAIT : {e}")
        import traceback
        traceback.print_exc()
        return None

# ─── EXÉCUTION ──────────────────────────────────────────────
json_files = sorted(JSON_DIR.glob('*.json'))

    for jf in json_files:
        try:
            data = json.load(open(jf, encoding='utf-8'))
            
            # Vérifier qu'il y a des données
            if not (data.get('ACTIF') or data.get('PASSIF') or data.get('TCR')):

                n_skip += 1

            result = generate_forfait(data)
            if result:
                n_gen += 1
            else:
                n_err += 1
                
        except Exception as e:
            print(f'❌ {jf.stem} : {e}')
            n_err += 1
    
    print(f'\n✅ {n_gen} FORFAIT générés | {n_skip} ignorés | {n_err} erreurs')
