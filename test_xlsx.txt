import re, shutil
from openpyxl.cell.cell import MergedCell

def _norm(s):
    return re.sub(r'[^a-z0-9]', '', str(s or '').lower())

def safe_write(ws, row, col, value):
    """Écrit en gérant les cellules fusionnées (écrit au haut-gauche de la plage)."""
    if value is None:
        return 0
    for mr in list(ws.merged_cells.ranges):
        if (mr.min_row <= row <= mr.max_row) and (mr.min_col <= col <= mr.max_col):
            ws.cell(row=mr.min_row, column=mr.min_col).value = value
            return 1
    ws.cell(row=row, column=col).value = value
    return 1

def to_kdzd(v):
    return None if v is None else round(v / 1000.0, 2)

# Colonnes ACTIF (déjà OK) : N brut=7 amort=8 net=9 ; N-1 brut=10 amort=11 net=12
ACTIF_COLS = {'brut_n':7, 'amort_n':8, 'net_n':9, 'brut_n1':10, 'amort_n1':11, 'net_n1':12}

# Colonnes PASSIF corrigées : N = E(5), N-1 = F(6)
PASSIF_COLS = {'n':5, 'n1':6}

# Colonnes TCR (déjà OK) : DEBIT N=6, CREDIT N=7, DEBIT N-1=8, CREDIT N-1=9
TCR_COLS = {'debit_n':6, 'credit_n':7, 'debit_n1':8, 'credit_n1':9}

def generate_forfait(result):
    if not TEMPLATE_FORFAIT.exists():
        print(f'❌ Template introuvable : {TEMPLATE_FORFAIT}'); return None
    ent = re.sub(r'[^A-Za-z0-9 _-]', '', result.get('entreprise') or 'ENTREPRISE').strip()
    annee = result.get('exercice') or 'XXXX'
    out = FORFAIT_OUTPUT_DIR / f'FORFAIT Cas 1_{ent}_{annee}.xlsx'
    shutil.copy2(TEMPLATE_FORFAIT, out)
    wb = load_workbook(out)
    a, p, t = result.get('ACTIF') or {}, result.get('PASSIF') or {}, result.get('TCR') or {}
    n_wr = 0

    # ── ACTIF ──
    if 'Saisie actif' in wb.sheetnames:
        ws = wb['Saisie actif']
        safe_write(ws, 2, 7, 'Fiscal')                      # Nature bilan
        if result.get('exercice'):
            safe_write(ws, 4, 7, result['exercice'])        # Date N
        for key, row in ACTIF_ROWS.items():
            for colkey, col in ACTIF_COLS.items():
                src = f'{key}_{colkey}'
                n_wr += safe_write(ws, row, col, to_kdzd(a.get(src)))

    # ── PASSIF (colonnes corrigées) ──
    if 'Saisie passif' in wb.sheetnames:
        ws = wb['Saisie passif']
        for key, row in PASSIF_ROWS.items():
            n_wr += safe_write(ws, row, PASSIF_COLS['n'],  to_kdzd(p.get(f'{key}_n')))
            n_wr += safe_write(ws, row, PASSIF_COLS['n1'], to_kdzd(p.get(f'{key}_n1')))

    # ── TCR ──
    if 'Saisie TCR' in wb.sheetnames:
        ws = wb['Saisie TCR']
        for key, row in TCR_ROWS.items():
            for colkey, col in TCR_COLS.items():
                n_wr += safe_write(ws, row, col, to_kdzd(t.get(f'{key}_{colkey}')))

    wb.save(out); wb.close()
    print(f'✅ {out.name} | {n_wr} cellules')
    return out
