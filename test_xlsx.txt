import re, shutil
from openpyxl import load_workbook

def norm(s):
    return re.sub(r'[^a-z0-9]', '', str(s or '').lower())

def kd(v):
    return None if v is None else round(v / 1000.0, 2)

def find_row(ws, label, max_col=4, max_row=90):
    t = norm(label)
    if not t:
        return None
    for r in range(1, max_row + 1):
        for c in range(1, max_col + 1):
            if norm(ws.cell(row=r, column=c).value) == t:
                return r
    for r in range(1, max_row + 1):
        for c in range(1, max_col + 1):
            v = norm(ws.cell(row=r, column=c).value)
            if v and (t in v or v in t):
                return r
    return None

def detect_actif_cols(ws):
    for r in range(1, 10):
        vals = {c: norm(ws.cell(row=r, column=c).value) for c in range(1, 16)}
        b = [c for c, v in vals.items() if 'montantsbruts' in v]
        a = [c for c, v in vals.items() if 'amortissements' in v]
        n = [c for c, v in vals.items() if v.startswith('net')]
        if b and a and n:
            if len(b) >= 3 and len(n) >= 3:
                return {'brut_n': b[1], 'amort_n': a[1], 'net_n': n[1], 'net_n1': n[2]}
            return {'brut_n': b[0], 'amort_n': a[0], 'net_n': n[0], 'net_n1': n[-1]}
    return None

def detect_passif_cols(ws):
    for r in range(1, 8):
        dates = [c for c in range(1, 10)
                 if re.search(r'\d{2}/\d{4}', str(ws.cell(row=r, column=c).value or ''))]
        if len(dates) >= 2:
            return {'n': dates[0], 'n1': dates[1]}
    return None

def detect_tcr_cols(ws):
    for r in range(1, 8):
        deb = [c for c in range(1, 12) if 'debit' in norm(ws.cell(row=r, column=c).value)]
        cred = [c for c in range(1, 12) if 'credit' in norm(ws.cell(row=r, column=c).value)]
        if len(deb) >= 2 and len(cred) >= 2:
            if len(deb) >= 3:
                return {'deb_n': deb[1], 'cred_n': cred[1], 'deb_n1': deb[2], 'cred_n1': cred[2]}
            return {'deb_n': deb[0], 'cred_n': cred[0], 'deb_n1': deb[1], 'cred_n1': cred[1]}
    return None

def generate_forfait(result):
    if not TEMPLATE_FORFAIT.exists():
        print('Template introuvable:', TEMPLATE_FORFAIT); return None
    ent = re.sub(r'[^A-Za-z0-9 _-]', '', result.get('entreprise') or 'ENTREPRISE').strip()
    annee = result.get('exercice') or 'XXXX'
    out = FORFAIT_OUTPUT_DIR / ('FORFAIT Cas 1_%s_%s.xlsx' % (ent, annee))
    shutil.copy2(TEMPLATE_FORFAIT, out)
    wb = load_workbook(out)
    a = result.get('ACTIF') or {}
    p = result.get('PASSIF') or {}
    t = result.get('TCR') or {}
    n_wr = 0

    if 'Saisie actif' in wb.sheetnames:
        ws = wb['Saisie actif']; cols = detect_actif_cols(ws)
        if cols:
            for key, lab in ACTIF_L.items():
                r = find_row(ws, lab)
                if not r: continue
                for ca, src in [('brut_n', key+'_brut'), ('amort_n', key+'_amort'),
                                ('net_n', key+'_n'), ('net_n1', key+'_n1')]:
                    v = kd(a.get(src))
                    if v is not None:
                        ws.cell(row=r, column=cols[ca], value=v); n_wr += 1

    if 'Saisie passif' in wb.sheetnames:
        ws = wb['Saisie passif']; cols = detect_passif_cols(ws)
        if cols:
            for key, lab in PASSIF_L.items():
                r = find_row(ws, lab)
                if not r: continue
                v = kd(p.get(key+'_n'))
                if v is not None: ws.cell(row=r, column=cols['n'], value=v); n_wr += 1
                v = kd(p.get(key+'_n1'))
                if v is not None: ws.cell(row=r, column=cols['n1'], value=v); n_wr += 1

    if 'Saisie TCR' in wb.sheetnames:
        ws = wb['Saisie TCR']; cols = detect_tcr_cols(ws)
        if cols:
            for key, lab in TCR_L.items():
                r = find_row(ws, lab)
                if not r: continue
                for ca, src in [('deb_n', key+'_debit_n'), ('cred_n', key+'_credit_n'),
                                ('deb_n1', key+'_debit_n1'), ('cred_n1', key+'_credit_n1')]:
                    v = kd(t.get(src))
                    if v is not None:
                        ws.cell(row=r, column=cols[ca], value=v); n_wr += 1

    wb.save(out); wb.close()
    print('FORFAIT %s : %d cellules' % (out.name, n_wr))
    return out
