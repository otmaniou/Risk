import shutil, re
from openpyxl import load_workbook
from openpyxl.styles import Font, PatternFill

RED     = Font(color='FFFF0000', bold=True)
REDFILL = PatternFill('solid', start_color='FFFFC7CE')

def to_kdzd(v):
    if v is None: return None
    return round(v / 1000.0, 2)

def _norm(s):
    return re.sub(r'[^a-z0-9]', '', str(s or '').lower())

def find_row(ws, label_col, key, max_row=80):
    target = _norm(key.replace('_', ' '))
    if not target: return None
    for r in range(1, min(ws.max_row, max_row) + 1):
        v = ws.cell(row=r, column=label_col).value
        if v is not None and target in _norm(v):
            return r
    return None

def _mark(c, rouge):
    if rouge:
        c.font = RED
        c.fill = REDFILL

def generate_forfait(result):
    if not TEMPLATE_FORFAIT.exists():
        print(f'❌ Template introuvable : {TEMPLATE_FORFAIT}'); return None

    entreprise = re.sub(r'[^A-Za-z0-9 _-]', '', result.get('entreprise') or 'ENTREPRISE').strip()
    annee = result.get('exercice') or 'XXXX'
    out_path = FORFAIT_OUTPUT_DIR / f'FORFAIT Cas 1_{entreprise}_{annee}.xlsx'

    shutil.copy2(TEMPLATE_FORFAIT, out_path)
    wb = load_workbook(out_path)

    a = result.get('ACTIF')  or {}
    p = result.get('PASSIF') or {}
    t = result.get('TCR')    or {}
    anoms = compute_controles(result)
    n_written = 0

    # ── ACTIF ──
    cfg = FORFAIT_COORDS['actif']
    if cfg['sheet'] in wb.sheetnames:
        ws = wb[cfg['sheet']]; cn, cn1 = cfg['cols_n'], cfg['cols_n1']
        for key, row in cfg['rows'].items():
            rr = find_row(ws, cfg['label_col'], key) or row
            rouge_n, rouge_n1 = f'ACTIF.{key}.n' in anoms, f'ACTIF.{key}.n1' in anoms
            for col, val, rouge in [
                (cn['brut'],  to_kdzd(a.get(f'{key}_brut')),  rouge_n),
                (cn['amort'], to_kdzd(a.get(f'{key}_amort')), rouge_n),
                (cn['net'],   to_kdzd(a.get(f'{key}_n')),     rouge_n),
                (cn1['net'],  to_kdzd(a.get(f'{key}_n1')),    rouge_n1),
            ]:
                if val is not None:
                    c = ws.cell(row=rr, column=col, value=val); n_written += 1
                    _mark(c, rouge)

    # ── PASSIF ──
    cfg = FORFAIT_COORDS['passif']
    if cfg['sheet'] in wb.sheetnames:
        ws = wb[cfg['sheet']]
        for key, row in cfg['rows'].items():
            rr = find_row(ws, cfg['label_col'], key) or row
            for col, suf in [(cfg['col_n'], 'n'), (cfg['col_n1'], 'n1')]:
                val = to_kdzd(p.get(f'{key}_{suf}'))
                if val is not None:
                    c = ws.cell(row=rr, column=col, value=val); n_written += 1
                    _mark(c, f'PASSIF.{key}.{suf}' in anoms)

    # ── TCR (ventilation DEBIT/CREDIT) ──
    cfg = FORFAIT_COORDS['tcr']
    if cfg['sheet'] in wb.sheetnames:
        ws = wb[cfg['sheet']]; ctn, ctn1 = cfg['cols_n'], cfg['cols_n1']
        for key, row in cfg['rows'].items():
            rr = find_row(ws, cfg['label_col'], key) or row
            for suf, cd, cc in [('n', ctn['debit'], ctn['credit']),
                                ('n1', ctn1['debit'], ctn1['credit'])]:
                d  = to_kdzd(t.get(f'{key}_debit_{suf}'))
                cr = to_kdzd(t.get(f'{key}_credit_{suf}'))
                rouge = f'TCR.{key}.{suf}' in anoms
                if d is not None:
                    c = ws.cell(row=rr, column=cd, value=d); n_written += 1
                    _mark(c, rouge)
                if cr is not None:
                    c = ws.cell(row=rr, column=cc, value=cr); n_written += 1
                    _mark(c, rouge)

    wb.save(out_path); wb.close()
    print(f'✅ {out_path.name} | {n_written} cellules | {len(anoms)} incohérences rouges')
    return out_path

print('✅ Générateur FORFAIT corrigé')
