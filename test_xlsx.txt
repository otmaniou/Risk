import re, shutil, json
from openpyxl import load_workbook

def _norm(s): return re.sub(r'[^a-z0-9]','',str(s or '').lower())

def to_kdzd(v):
    if v is None: return None
    return round(v/1000.0, 2)

def _clean_name(s):
    return re.sub(r'[^A-Za-z0-9 _-]','', s or 'ENTREPRISE').strip()

def find_row(ws, label, max_row=90, cols=(1,2,3)):
    t=_norm(label)
    if not t: return None
    for r in range(1, max_row+1):
        for c in cols:
            v=ws.cell(row=r, column=c).value
            if v is not None and t in _norm(v): return r
    return None

def detect_actif_cols(ws):
    for r in range(1,12):
        brut=[c for c in range(1,20) if ws.cell(row=r,column=c).value and 'montantsbruts' in _norm(ws.cell(row=r,column=c).value)]
        if len(brut)>=2:
            b,b1=brut[0],brut[1]
            return dict(brut_n=b,amort_n=b+1,net_n=b+2,brut_n1=b1,amort_n1=b1+1,net_n1=b1+2)
        if len(brut)==1:
            b=brut[0]
            return dict(brut_n=b,amort_n=b+1,net_n=b+2,brut_n1=b+3,amort_n1=b+4,net_n1=b+5)
    return dict(brut_n=2,amort_n=3,net_n=4,brut_n1=5,amort_n1=6,net_n1=7)

def detect_passif_cols(ws):
    for r in range(1,8):
        ys=[c for c in range(1,10) if ws.cell(row=r,column=c).value and re.search(r'\d{2}/\d{4}',str(ws.cell(row=r,column=c).value))]
        if len(ys)>=2: return ys[0], ys[1]
    return 2,3

def detect_tcr_cols(ws):
    for r in range(1,8):
        deb=[c for c in range(1,12) if ws.cell(row=r,column=c).value and 'debit' in _norm(ws.cell(row=r,column=c).value)]
        cred=[c for c in range(1,12) if ws.cell(row=r,column=c).value and 'credit' in _norm(ws.cell(row=r,column=c).value)]
        if deb and cred:
            return dict(deb_n=deb[0],cred_n=cred[0],
                        deb_n1=deb[1] if len(deb)>1 else deb[0]+2,
                        cred_n1=cred[1] if len(cred)>1 else cred[0]+2)
    return dict(deb_n=3,cred_n=4,deb_n1=5,cred_n1=6)

def generate_forfait(result):
    if not TEMPLATE_FORFAIT.exists():
        print(f'❌ Template introuvable : {TEMPLATE_FORFAIT}'); return None
    ent=_clean_name(result.get('entreprise')); annee=result.get('exercice') or 'XXXX'
    out=FORFAIT_OUTPUT_DIR / f'FORFAIT Cas 1_{ent}_{annee}.xlsx'
    shutil.copy2(TEMPLATE_FORFAIT, out)
    wb=load_workbook(out)
    a=result.get('ACTIF') or {}; p=result.get('PASSIF') or {}; t=result.get('TCR') or {}
    n_written=0

    # ── ACTIF ──
    if 'Saisie actif' in wb.sheetnames:
        ws=wb['Saisie actif']; cols=detect_actif_cols(ws)
        for key,label in SCHEMAS['ACTIF']['postes'].items():
            r=find_row(ws,label)
            if not r: continue
            for ca,jk in [('brut_n','_brut'),('amort_n','_amort'),('net_n','_n'),('net_n1','_n1')]:
                v=to_kdzd(a.get(key+jk))
                if v is not None: ws.cell(row=r,column=cols[ca],value=v); n_written+=1

    # ── PASSIF ──
    if 'Saisie passif' in wb.sheetnames:
        ws=wb['Saisie passif']; cn,cn1=detect_passif_cols(ws)
        for key,label in SCHEMAS['PASSIF']['postes'].items():
            r=find_row(ws,label)
            if not r: continue
            v=to_kdzd(p.get(key+'_n'))
            if v is not None: ws.cell(row=r,column=cn,value=v); n_written+=1
            v=to_kdzd(p.get(key+'_n1'))
            if v is not None: ws.cell(row=r,column=cn1,value=v); n_written+=1

    # ── TCR ──
    if 'Saisie TCR' in wb.sheetnames:
        ws=wb['Saisie TCR']; cols=detect_tcr_cols(ws)
        for key,label in SCHEMAS['TCR']['postes'].items():
            r=find_row(ws,label,cols=(1,2))
            if not r: continue
            for jk,col in [('_debit_n',cols['deb_n']),('_credit_n',cols['cred_n']),
                           ('_debit_n1',cols['deb_n1']),('_credit_n1',cols['cred_n1'])]:
                v=to_kdzd(t.get(key+jk))
                if v is not None: ws.cell(row=r,column=col,value=v); n_written+=1

    wb.save(out); wb.close()
    print(f'✅ {out.name} | {n_written} cellules remplies')
    return out

# ── Boucle ──
n_gen=0
for jf in sorted(JSON_DIR.glob('*.json')):
    try:
        d=json.load(open(jf,encoding='utf-8'))
        if not (d.get('ACTIF') or d.get('PASSIF') or d.get('TCR')): continue
        if generate_forfait(d): n_gen+=1
    except Exception as e:
        print(f'❌ {jf.stem}: {e}')
print(f'✅ {n_gen} FORFAIT .xlsx générés → {FORFAIT_OUTPUT_DIR}')
