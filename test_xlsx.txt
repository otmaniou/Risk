import shutil, re
from openpyxl import load_workbook

def to_kdzd(v):
    return None if v is None else round(v/1000.0, 2)

def _clean(s):
    return re.sub(r'[^A-Za-z0-9 _-]','', s or 'ENTREPRISE').strip()

def _norm(s):
    s = re.sub(r'[^a-z0-9]','', str(s or '').lower())
    return s

def build_row_map(ws, label_cols):
    """Map normalized_label -> row_index (une seule fois, template figé)."""
    m = {}
    for r in range(1, ws.max_row+1):
        for c in label_cols:
            v = ws.cell(row=r, column=c).value
            if v is not None:
                m.setdefault(_norm(v), r)
    return m
