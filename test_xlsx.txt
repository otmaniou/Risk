import openpyxl, json

print('--- 1) fichiers xlsx produits (heure + taille) ---')
for f in sorted(XLSX_DIR.glob('*.xlsx')):
    import datetime as _dt
    print(f.name, '|', _dt.datetime.fromtimestamp(f.stat().st_mtime), '|', f.stat().st_size, 'o')

print()
print('--- 2) resume de la transposition du dernier dossier ---')
dernier = sorted(CONTROLE_DIR.glob('*.controle.json'))[-1]
doc = json.loads(dernier.read_text(encoding='utf-8'))
t = doc.get('controles', {}).get('transposition', {})
print('classeur      :', t.get('classeur'))
print('valeurs_ecrites:', t.get('valeurs_ecrites'))
print('par_tableau   :', t.get('valeurs_par_tableau'))

print()
print('--- 3) ce que la boucle devrait ecrire (8 premiers) ---')
fixes, dyn = _postes_du_document(doc)
print('total fixes attendus :', len(fixes), '| dynamiques :', {k: len(v) for k, v in dyn.items()})
for tab, rc, col, val in fixes[:8]:
    li = MAP_LIGNES.get(tab, {}).get(rc)
    co = MAP_COLONNES.get(tab, {}).get(col)
    print(f'  {tab} {rc} {col} -> {co}{li[0]} = {val}')

print()
print('--- 4) contenu reel du xlsx sorti (1. Bilan Actif, lignes 11-15) ---')
out = sorted(XLSX_DIR.glob('*.xlsx'))[-1]
wb = openpyxl.load_workbook(out)
ws = wb['1. Bilan Actif']

    print(r, [ws.cell(r, c).value for c in range(1, 6)])
