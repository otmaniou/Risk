# CELLULE 15bis — AUTO-CALIBRATION DU MODELE | BILANS_V11A
# Objectif: reconstruire MAP_LIGNES / MAP_DYNAMIQUES en lisant les
#           libellés imprimés du modèle réel (_5.xlsx), pour absorber
#           tout décalage d'en-tête entre versions du classeur.
# Entrées: MODELE_XLSX, SCHEMAS, MAP_FEUILLES, MAP_LIGNES, MAP_DYNAMIQUES.
# Sorties: MAP_LIGNES / MAP_DYNAMIQUES recalés + affichage de contrôle.
# Règles: on conserve le drapeau "calculée par formule" de l'ancien mapping ;
#         un tableau n'est recalé que si ≥ 60 % de ses libellés sont retrouvés.
# ════════════════════════════════════════════════════════════
def _nl(s):
    if s is None:
        return ''
    s = ''.join(ch for ch in unicodedata.normalize('NFKD', str(s))
                if unicodedata.category(ch) != 'Mn')
    return re.sub(r'[^a-z0-9]+', ' ', s.lower()).strip()

def _index_labels(ws, max_row=220):
    idx = {}
    for r in range(1, min(ws.max_row, max_row) + 1):
        for c in ('A', 'B'):
            v = ws[c + str(r)].value
            if isinstance(v, str) and v.strip():
                idx.setdefault(_nl(v), r)
    return idx

def calibrer_modele(chemin_modele=MODELE_XLSX):
    wb = openpyxl.load_workbook(chemin_modele, data_only=False)
    for tab, feuille in MAP_FEUILLES.items():
        if feuille not in wb.sheetnames:
            continue
        ws = wb[feuille]
        idx = _index_labels(ws)

        # ── tableaux fixes : ligne = première occurrence du libellé imprimé ──
        new = {}
        for rc, lab in SCHEMAS[tab]['postes'].items():
            nl = _nl(lab)
            if not nl:
                continue
            hit = None
            for text, r in idx.items():
                if (nl in text) or (len(text) >= 6 and text in nl):
                    hit = r
                    break
            if hit:
                drapeau_formule = MAP_LIGNES.get(tab, {}).get(rc, (hit, False))[1]
                new[rc] = (hit, drapeau_formule)
        if len(new) >= 0.6 * len(SCHEMAS[tab]['postes']):
            MAP_LIGNES[tab] = new

        # ── tableaux dynamiques : plage = entre l'en-tête et le premier TOTAL ──
        if tab in MAP_DYNAMIQUES:
            debut, fin, col_lib = MAP_DYNAMIQUES[tab]
            r_tot = next((r for text, r in idx.items() if text.startswith('total')), None)
            if r_tot and r_tot > debut:
                MAP_DYNAMIQUES[tab] = (min(debut, r_tot - 1), r_tot - 1, col_lib)
    wb.close()

calibrer_modele()

print('✅ Modèle calibré sur les libellés imprimés')
for tab, rc in [('ACTIF', 'terrains'), ('ACTIF', 'total_general_actif'),
                ('PASSIF', 'capital_emis'), ('TCR', 'charges_personnel'),
                ('A9', 'total_reintegrations'), ('A10', 'affectation_dividendes')]:
    print('  ', tab, rc, '-> ligne', MAP_LIGNES.get(tab, {}).get(rc))
