import traceback

NOM_PDF = 'Bilan MOF PHARMA 2025.pdf'
pdf_path = INPUT_DIR / NOM_PDF

print('fichier:', pdf_path, '| existe:', pdf_path.exists(), '| taille:', pdf_path.stat().st_size if pdf_path.exists() else 0)

# Étape 1 : ouverture PDF
try:
    doc = fitz.open(pdf_path)
    print('ETAPE 1 OK | ouverture | nb pages:', len(doc), '| encrypte:', doc.is_encrypted)
    doc.close()
except Exception as e:
    print('ETAPE 1 ECHEC | ouverture |', type(e).__name__, repr(e))

# Étape 2 : rendu + deskew + blank
try:
    pages = pdf_to_pages(pdf_path)
    print('ETAPE 2 OK | rendu | pages:', len(pages))
    actives = [p for p in pages if not is_blank(p['image'])]
    print('ETAPE 2 | pages actives:', len(actives), '| blanches:', len(pages) - len(actives))
except Exception:
    print('ETAPE 2 ECHEC | rendu')
    traceback.print_exc()
    actives = []

# Étape 3 : classification VLM
if actives:
    try:
        mini = [resize(p['image'], 600) for p in actives]
        reps = ask_batch(PROMPT_CLASSIF, mini)
        print('ETAPE 3 OK | classification |', [r['text'][:30] for r in reps])
    except Exception:
